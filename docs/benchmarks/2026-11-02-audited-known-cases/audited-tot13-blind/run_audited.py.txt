"""Run one audited known-case master batch with the last ToT13 settings."""

import json
import os
from pathlib import Path

from pydantic import SecretStr

from capstone.experiment import main
from capstone.settings import Settings

ROOT = Path(__file__).resolve().parent
OUT = ROOT / "docs/benchmarks/2026-11-02-audited-known-cases/audited-tot13-blind"
NAME = "master-audited-20261102-tot13-blind"

fixed = {
    "DEFAULT_MODEL": "nvidia/nvidia/nemotron-3.5-super-vl-120b-a12b",
    "FALLBACK_MODEL": "cohere/command-a-03-2025",
    "CRITIC_MODEL": "mistral/ministral-14b-2512",
    "CRITIC_FALLBACK_MODEL": "cohere/command-a-03-2025",
    "CRITIC": True,
    "CRITIC_BLIND": True,
    "CRITIC_FOLLOWUP": True,
    "CRITIC_SECONDS": 40,
    "CRITIC_FOLLOWUP_SECONDS": 90,
    "MANUAL_STEP": True,
    "STEP_QUESTIONS": 1,
    "CHROMA_PATH": str(ROOT / ".local-audited-manual-index"),
    "MEMORY_RETRIEVAL_MODE": "none",
    "EMBEDDING_MODEL": "nvidia/nvidia/nemotron-3-embed-1b",
    "PROVIDER_SAMPLING": {},
    "MODEL_SAMPLING": {
        "nvidia/nvidia/nemotron-3.5-super-vl-120b-a12b": {"temperature": 1.0, "top_p": 0.95},
        "cohere/command-a-03-2025": {"temperature": 0.3, "top_p": 0.75},
        "mistral/ministral-14b-2512": {"temperature": 0.0, "top_p": None},
    },
}
for job in ("PLAN", "DECIDE", "INTERPRET", "HYPOTHESIZE", "CONCLUDE", "JUDGE", "LESSON"):
    fixed[f"{job}_MODEL"] = None

settings = Settings(_env_file=ROOT.parent.parent / ".env", **fixed)
for field in ("nvidia_api_key", "cohere_api_key", "mistral_api_key", "typesafe_api_key"):
    if getattr(settings, field) is None:
        raise RuntimeError(f"Missing {field.upper()}")
private = {name for name in Settings.model_fields if name.endswith(("_api_key", "_api_token")) or name in ("database_url", "cloudflare_account_id")}
public = settings.model_dump(mode="json", exclude=private)
for name, field in Settings.model_fields.items():
    alias = str(field.validation_alias or name)
    value = getattr(settings, name)
    os.environ.pop(alias, None)
    if value is None:
        continue
    if isinstance(value, SecretStr):
        os.environ[alias] = value.get_secret_value()
    elif isinstance(value, bool):
        os.environ[alias] = "on" if value else "off"
    elif isinstance(value, Path):
        os.environ[alias] = str(value)
    elif name in ("provider_sampling", "model_sampling"):
        os.environ[alias] = json.dumps(public[name])
    else:
        os.environ[alias] = str(value)

OUT.mkdir(parents=True, exist_ok=True)
if (OUT / f"{NAME}.records.jsonl").exists():
    raise RuntimeError("Records already exist; refusing to overwrite")
raise SystemExit(main([
    "run", "--name", NAME, "--known", "--tree-of-thought", "--manual", "semantic",
    "--approve", "all", "--out", str(OUT / f"{NAME}.records.jsonl"),
    "--audit", str(OUT / f"{NAME}.audit.jsonl"), "--quiet",
]))
